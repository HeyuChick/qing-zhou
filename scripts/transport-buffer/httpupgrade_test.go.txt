type qzConn struct {
	input   *bytes.Reader
	written bytes.Buffer
	limit   int
}

func (c *qzConn) Read(p []byte) (int, error) {
	if c.limit > 0 && len(p) > c.limit {
		p = p[:c.limit]
	}
	return c.input.Read(p)
}
func (c *qzConn) Write(p []byte) (int, error)      { return c.written.Write(p) }
func (c *qzConn) Close() error                     { return nil }
func (c *qzConn) LocalAddr() net.Addr              { return &net.TCPAddr{IP: net.IPv4(127, 0, 0, 1), Port: 1234} }
func (c *qzConn) RemoteAddr() net.Addr             { return &net.TCPAddr{IP: net.IPv4(127, 0, 0, 1), Port: 4321} }
func (c *qzConn) SetDeadline(time.Time) error      { return nil }
func (c *qzConn) SetReadDeadline(time.Time) error  { return nil }
func (c *qzConn) SetWriteDeadline(time.Time) error { return nil }

type qzDialer struct{ conn net.Conn }

func (d qzDialer) DialContext(context.Context, string, M.Socksaddr) (net.Conn, error) {
	return d.conn, nil
}
func (d qzDialer) ListenPacket(context.Context, M.Socksaddr) (net.PacketConn, error) {
	return nil, errors.New("not used")
}

func TestQZHTTPUpgradeClientPreservesCoalescedFirstPayload(t *testing.T) {
	payload := []byte{0, 146, 1, 2, 3, 4, 5, 6, 7, 8, 9}
	wire := append([]byte("HTTP/1.1 101 Switching Protocols\r\nConnection: Upgrade\r\nUpgrade: websocket\r\n\r\n"), payload...)
	raw := &qzConn{input: bytes.NewReader(wire)}
	client, err := NewClient(context.Background(), qzDialer{raw}, M.ParseSocksaddr("127.0.0.1:1234"), option.V2RayHTTPUpgradeOptions{Path: "/fixture"}, nil)
	if err != nil {
		t.Fatal(err)
	}
	conn, err := client.DialContext(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	got, err := io.ReadAll(conn)
	if err != nil {
		t.Fatal(err)
	}
	if !bytes.Equal(got, payload) {
		t.Fatalf("first protocol bytes lost after HTTP 101: got %x want %x", got, payload)
	}
}

type qzHijacker struct {
	conn   *qzConn
	reader *bufio.Reader
	header http.Header
	status int
}

func (w *qzHijacker) Header() http.Header         { return w.header }
func (w *qzHijacker) Write(p []byte) (int, error) { return len(p), nil }
func (w *qzHijacker) WriteHeader(status int)      { w.status = status }
func (w *qzHijacker) FlushError() error           { return nil }
func (w *qzHijacker) Hijack() (net.Conn, *bufio.ReadWriter, error) {
	return w.conn, bufio.NewReadWriter(w.reader, bufio.NewWriter(io.Discard)), nil
}

type qzHandler struct {
	payload []byte
	err     error
}

func (h *qzHandler) NewConnectionEx(ctx context.Context, conn net.Conn, source, destination M.Socksaddr, onClose N.CloseHandlerFunc) {
	h.payload, h.err = io.ReadAll(conn)
}

func TestQZHTTPUpgradeServerPreservesHijackReadAhead(t *testing.T) {
	for _, n := range []int{1, 4, 11} {
		t.Run(fmt.Sprint(n), func(t *testing.T) {
			payload := []byte{0, 146, 1, 2, 3, 4, 5, 6, 7, 8, 9}
			raw := &qzConn{input: bytes.NewReader(payload), limit: n}
			reader := bufio.NewReader(raw)
			if _, err := reader.Peek(n); err != nil {
				t.Fatal(err)
			}
			if reader.Buffered() != n {
				t.Fatalf("fixture requires exactly %d bytes of read-ahead; got %d", n, reader.Buffered())
			}
			raw.limit = 0
			writer := &qzHijacker{conn: raw, reader: reader, header: make(http.Header)}
			handler := new(qzHandler)
			server := &Server{path: "/fixture", handler: handler}
			request := &http.Request{Method: http.MethodGet, URL: &url.URL{Path: "/fixture"}, Host: "localhost", Header: make(http.Header)}
			request.Header.Set("Connection", "Upgrade")
			request.Header.Set("Upgrade", "websocket")
			server.ServeHTTP(writer, request)
			if writer.status != 101 || handler.err != nil {
				t.Fatalf("handshake failed status=%d err=%v", writer.status, handler.err)
			}
			if !bytes.Equal(handler.payload, payload) {
				t.Fatalf("%d pre-read bytes lost by Hijack: got %x want %x", n, handler.payload, payload)
			}
		})
	}
}

func TestQZHTTPUpgradeFragmentedAndLongFlow(t *testing.T) {
	for _, limit := range []int{1, 7, 0} {
		for _, size := range []int{0, 1, 65536, 1048576} {
			t.Run(fmt.Sprintf("chunk-%d/bytes-%d", limit, size), func(t *testing.T) {
				payload := bytes.Repeat([]byte{0x53}, size)
				wire := append([]byte("HTTP/1.1 101 Switching Protocols\r\nConnection: Upgrade\r\nUpgrade: websocket\r\n\r\n"), payload...)
				raw := &qzConn{input: bytes.NewReader(wire), limit: limit}
				client, err := NewClient(context.Background(), qzDialer{raw}, M.ParseSocksaddr("127.0.0.1:1234"), option.V2RayHTTPUpgradeOptions{Path: "/fixture"}, nil)
				if err != nil {
					t.Fatal(err)
				}
				conn, err := client.DialContext(context.Background())
				if err != nil {
					t.Fatal(err)
				}
				got, err := io.ReadAll(conn)
				if err != nil || !bytes.Equal(got, payload) {
					t.Fatalf("flow truncated or altered size=%d got=%d err=%v", size, len(got), err)
				}
				if err = conn.Close(); err != nil {
					t.Fatal(err)
				}
			})
		}
	}
}

func TestQZHTTPUpgradeConcurrentIndependentBuffers(t *testing.T) {
	var wg sync.WaitGroup
	for i := 0; i < 32; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			payload := bytes.Repeat([]byte{byte(i + 1)}, 4096+i)
			raw := &qzConn{input: bytes.NewReader(payload), limit: 1}
			reader := bufio.NewReader(raw)
			if _, err := reader.Peek(1); err != nil {
				t.Error(err)
				return
			}
			raw.limit = 0
			writer := &qzHijacker{conn: raw, reader: reader, header: make(http.Header)}
			handler := new(qzHandler)
			server := &Server{path: "/fixture", handler: handler}
			request := &http.Request{Method: http.MethodGet, URL: &url.URL{Path: "/fixture"}, Host: "localhost", Header: make(http.Header)}
			request.Header.Set("Connection", "Upgrade")
			request.Header.Set("Upgrade", "websocket")
			server.ServeHTTP(writer, request)
			if handler.err != nil || !bytes.Equal(handler.payload, payload) {
				t.Errorf("concurrent buffer mixed/lost user=%d got=%d err=%v", i, len(handler.payload), handler.err)
			}
		}(i)
	}
	wg.Wait()
}
